org 0x7E00
bits 16

%define KERNEL_LOAD_SEG 0x1000
%define KERNEL_START_LBA 34
%define KERNEL_SECTORS   9

start:
    cli

    ; setup segments
    xor ax, ax
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7C00

    mov si, msg_hello
    call puts

    ; enable A20
    mov ax, 0x2401
    int 0x15

    ; ----------------------------
    ; load kernel via int13
    ; ----------------------------
    mov word [lba], KERNEL_START_LBA
    mov word [left], KERNEL_SECTORS

    mov ax, KERNEL_LOAD_SEG
    mov es, ax
    xor bx, bx

.load:
    mov ah, 0x02
    mov al, 1
    mov ch, 0
    mov cl, byte [lba]
    inc cl
    mov dh, 0
    mov dl, 0x80

    int 0x13
    jc disk_fail

    add bx, 512
    inc word [lba]
    dec word [left]
    jnz .load

    mov si, msg_ok
    call puts

; ----------------------------
; GDT + PM switch
; ----------------------------
switch_pm:
    cli

    lgdt [gdt_desc]

    mov eax, cr0
    or eax, 1
    mov cr0, eax

    jmp 0x08:pm_entry

disk_fail:
    mov si, msg_fail
    call puts
    hlt

; ----------------------------
; real mode print
; ----------------------------
puts:
    lodsb
    or al, al
    jz .done
    mov ah, 0x0E
    int 0x10
    jmp puts
.done:
    ret

; ----------------------------
; data
; ----------------------------
lba:  dw 0
left: dw 0

msg_hello db "Stage 2 booting...",0
msg_ok    db "Kernel loaded",0
msg_fail  db "Disk error",0

; ----------------------------
; GDT
; ----------------------------
gdt_start:
    dq 0x0000000000000000

gdt_code:
    dw 0xFFFF
    dw 0x0000
    db 0x00
    db 10011010b
    db 11001111b
    db 0x00

gdt_data:
    dw 0xFFFF
    dw 0x0000
    db 0x00
    db 10010010b
    db 11001111b
    db 0x00

gdt_end:

gdt_desc:
    dw gdt_end - gdt_start - 1
    dd gdt_start

; ----------------------------
; protected mode
; ----------------------------
bits 32
pm_entry:
    mov ax, 0x10
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov fs, ax
    mov gs, ax

    mov esp, 0x90000

    ; write something to screen
    mov dword [0xB8000], 0x2F4B2F4F

    jmp 0x08:0x00100000

hang:
    cli
    hlt
    jmp hang
