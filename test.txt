org 0x0
bits 16

%define ENDL 0x0D, 0x0A

start:
    mov ax, 0x2000
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7000

    mov si, msg_hello
    call puts

    ; enable A20
    mov ax, 0x2401
    int 0x15

    ; load kernel one sector at a time
    mov word [curr_lba], 34
    mov word [sects_left], 9
    mov ax, 0x1000
    mov es, ax
    mov bx, 0

.load_loop:
    mov ax, [curr_lba]
    call lba_to_chs
    mov ah, 0x02
    mov al, 1
    int 0x13
    jc .disk_error
    add bx, 512
    inc word [curr_lba]
    dec word [sects_left]
    jnz .load_loop

    mov si, msg_kernel_ok
    call puts
    jmp .do_pm

.disk_error:
    mov si, msg_kernel_error
    call puts
    cli
    hlt

.do_pm:
    cli
    xor eax, eax
    mov ax, ds
    shl eax, 4
    add eax, gdt_start
    mov [gdt_descriptor + 2], eax
    lgdt [gdt_descriptor]
    mov eax, cr0
    or eax, 1
    mov cr0, eax
    db 0x66, 0xEA
    dd protected_mode + 0x20000
    dw 0x08

lba_to_chs:
    push ax
    push dx
    xor dx, dx
    mov bx, 18
    div bx
    inc dx
    mov cl, dl
    xor dx, dx
    mov bx, 2
    div bx
    mov dh, dl
    mov ch, al
    mov dl, 0
    pop ax
    mov dl, al
    pop ax
    ret

puts:
    push si
    push ax
.loop:
    lodsb
    or al, al
    jz .done
    mov ah, 0x0e
    mov bh, 0
    int 0x10
    jmp .loop
.done:
    pop ax
    pop si
    ret

curr_lba: dw 0
sects_left: dw 0

gdt_start:
    dq 0
gdt_code:
    dw 0xFFFF
    dw 0x0000
    db 0x00
    db 10011010b
    db 11001111b
    db 0x00
gdt_data:
    dw 0xFFFF
    dw 0x0000
    db 0x00
    db 10010010b
    db 11001111b
    db 0x00
gdt_end:

gdt_descriptor:
    dw gdt_end - gdt_start - 1
    dd 0

bits 32
protected_mode:
    mov ax, 0x10
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax
    mov esp, 0x90000
    mov dword [0xB8000], 0x2F4F2F4B
    mov byte [0xB8004], 'J'
    mov byte [0xB8005], 0x0F
    jmp 0x10000
    cli
    hlt

msg_hello: db 'Stage 2 loaded, entering protected mode...', ENDL, 0
msg_kernel_ok: db 'Kernel loaded!', ENDL, 0
msg_kernel_error: db 'Kernel load failed!', ENDL, 0
