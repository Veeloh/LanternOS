# 1. The ACPI hardware ID Linux matched (this is the real identifier, e.g. ELAN0634)
cat /sys/bus/acpi/devices/*/hid 2>/dev/null | grep -i elan
# or, more reliably:
grep -i elan /sys/bus/acpi/devices/*/hid 2>/dev/null

# 2. Confirms the same thing plus driver binding from dmesg
dmesg | grep -i elan

# 3. Full input device entry — shows the reported name/vendor/product IDs
cat /proc/bus/input/devices | grep -B2 -A10 -i elan

# 4. Kernel driver module info (confirms which driver bound, elan_i2c vs elantech vs i2c_hid)
lsmod | grep -i elan

# 5. If elan_i2c bound, it exposes real device metadata in sysfs — this is gold,
#    it gives you actual product_id / firmware version / ic_type straight from the chip
for f in /sys/bus/i2c/devices/i2c-ELAN*/; do
  echo "== $f =="
  cat "$f"/name 2>/dev/null
  cat "$f"/firmware_version 2>/dev/null
  cat "$f"/product_id 2>/dev/null
  cat "$f"/sm_version 2>/dev/null
  cat "$f"/iap_version 2>/dev/null
done

# 6. The actual I2C bus + address it's sitting on (cross-check against your 0x15 addr assumption)
i2cdetect -l
cat /sys/bus/i2c/devices/i2c-ELAN*/name 2>/dev/null
